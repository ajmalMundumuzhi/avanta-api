require("dotenv").config();

const mongoose = require("mongoose");
const bcrypt = require("bcrypt");
const readline = require("readline");
const AdminUser = require("../src/models/AdminUser");

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

const ask = (question) =>
  new Promise((resolve) => rl.question(question, resolve));

async function createAdmin() {
  try {
    const name = (await ask("Admin name: ")).trim();
    const email = (await ask("Admin email: ")).trim().toLowerCase();
    const password = await ask("Admin password: ");

    if (!name || !email || password.length < 12) {
      throw new Error(
        "Name and email are required. Password must be at least 12 characters."
      );
    }

    if (!process.env.MONGODB_URI) {
      throw new Error("MONGODB_URI is missing from .env");
    }

    await mongoose.connect(process.env.MONGODB_URI);

    const existing = await AdminUser.findOne({ email });

    if (existing) {
      throw new Error("An admin with this email already exists.");
    }

    const passwordHash = await bcrypt.hash(password, 12);

    const admin = await AdminUser.create({
      name,
      email,
      passwordHash,
      role: "admin",
      isActive: true,
    });

    console.log(`Admin created successfully: ${admin.email}`);
  } catch (error) {
    console.error("Admin creation failed:", error.message);
    process.exitCode = 1;
  } finally {
    rl.close();
    await mongoose.disconnect().catch(() => {});
  }
}

createAdmin();

      "MONGODB_URI is not configured. Starting without a database connection."
    );
    return false;

    console.log("MongoDB connected successfully.");
    return true;

    console.error("MongoDB connection failed:", error.message);
    throw error;
  }

  const states = {
    0: "disconnected",
    1: "connected",
    2: "connecting",
    3: "disconnecting",
  };

  return states[mongoose.connection.readyState] || "unknown";
}

  connectDatabase,
  getDatabaseStatus,
};

const nodemailer = require("nodemailer");

const smtpConfigured = Boolean(
  process.env.SMTP_HOST &&
  process.env.SMTP_USER &&
  process.env.SMTP_PASS
);

const transporter = smtpConfigured
  ? nodemailer.createTransport({
      host: process.env.SMTP_HOST,
      port: Number(process.env.SMTP_PORT || 465),
      secure:
        process.env.SMTP_SECURE === "true" ||
        Number(process.env.SMTP_PORT || 465) === 465,
      auth: {
        user: process.env.SMTP_USER,
        pass: process.env.SMTP_PASS,
      },
      connectionTimeout: 10000,
      greetingTimeout: 10000,
      socketTimeout: 15000,
    })
  : null;

async function sendEmail({ to, subject, text, html, replyTo }) {
  if (!transporter) {
    throw new Error("SMTP is not configured.");
  }

  return transporter.sendMail({
    from: `"Avanta Living" <${process.env.SMTP_USER}>`,
    to,
    subject,
    text,
    html,
    ...(replyTo ? { replyTo } : {}),
  });
}

async function verifyEmailConnection() {
  if (!transporter) {
    throw new Error("SMTP is not configured.");
  }

  return transporter.verify();
}

module.exports = { sendEmail, verifyEmailConnection };

const bcrypt = require("bcrypt");
const jwt = require("jsonwebtoken");
const AdminUser = require("../models/AdminUser");

exports.login = async (req, res, next) => {
  try {
    const { email, password } = req.body;

    if (
      typeof email !== "string" ||
      typeof password !== "string" ||
      !email.trim() ||
      !password
    ) {
      return res.status(400).json({
        success: false,
        error: {
          code: "VALIDATION_ERROR",
          message: "Email and password are required.",
        },
      });
    }

    const admin = await AdminUser.findOne({
      email: email.trim().toLowerCase(),
      isActive: true,
    }).select("+passwordHash");

    if (!admin || !(await bcrypt.compare(password, admin.passwordHash))) {
      return res.status(401).json({
        success: false,
        error: {
          code: "INVALID_CREDENTIALS",
          message: "Invalid email or password.",
        },
      });
    }

    if (!process.env.ADMIN_JWT_SECRET) {
      throw new Error("ADMIN_JWT_SECRET is not configured.");
    }

    const token = jwt.sign(
      {
        sub: admin._id.toString(),
        role: admin.role,
      },
      process.env.ADMIN_JWT_SECRET,
      {
        expiresIn: "8h",
        issuer: "avanta-living-api",
        audience: "avanta-living-admin",
      }
    );

    admin.lastLogin = new Date();
    await admin.save();

    return res.status(200).json({
      success: true,
      data: {
        token,
        expiresIn: 28800,
        admin: {
          id: admin._id,
          name: admin.name,
          email: admin.email,
          role: admin.role,
        },
      },
    });
  } catch (error) {
    next(error);
  }
};

const mongoose = require("mongoose");
const Property = require("../models/Property");
const Booking = require("../models/Booking");
const BlockedDate = require("../models/BlockedDate");

exports.getAvailability = async (req, res, next) => {
  try {
    const { propertyId, start, end } = req.query;

    if (!mongoose.isValidObjectId(propertyId)) {
      return res.status(400).json({
        success: false,
        error: { code: "INVALID_PROPERTY_ID", message: "Valid propertyId is required." },
      });
    }

    const property = await Property.findById(propertyId).select("propertyId title").lean();

    if (!property) {
      return res.status(404).json({
        success: false,
        error: { code: "PROPERTY_NOT_FOUND", message: "Property not found." },
      });
    }

    const startDate = start ? new Date(start) : new Date();
    const endDate = end ? new Date(end) : new Date(startDate.getTime() + 90 * 86400000);

    if (
      Number.isNaN(startDate.getTime()) ||
      Number.isNaN(endDate.getTime()) ||
      startDate >= endDate
    ) {
      return res.status(400).json({
        success: false,
        error: { code: "INVALID_DATE_RANGE", message: "Provide a valid start and end date." },
      });
    }

    const [bookings, blockedDates] = await Promise.all([
      Booking.find({
        property: property._id,
        status: "confirmed",
        checkin: { $lt: endDate },
        checkout: { $gt: startDate },
      }).select("reference checkin checkout status").lean(),
      BlockedDate.find({
        property: property._id,
        startDate: { $lt: endDate },
        endDate: { $gt: startDate },
      }).lean(),
    ]);

    res.json({
      success: true,
      data: { property, bookings, blockedDates },
    });
  } catch (error) {
    next(error);
  }
};


exports.createBlock = async (req, res, next) => {
  try {
    const { propertyId, startDate, endDate, type, source, notes } = req.body;

    if (!propertyId || !startDate || !endDate) {
      return res.status(400).json({
        success: false,
        error: {
          code: "VALIDATION_ERROR",
          message: "Property, start date and end date are required.",
        },
      });
    }

    if (!mongoose.isValidObjectId(propertyId)) {
      return res.status(400).json({
        success: false,
        error: {
          code: "INVALID_PROPERTY_ID",
          message: "Invalid property ID.",
        },
      });
    }

    const start = new Date(startDate);
    const end = new Date(endDate);

    if (
      Number.isNaN(start.getTime()) ||
      Number.isNaN(end.getTime()) ||
      start >= end
    ) {
      return res.status(400).json({
        success: false,
        error: {
          code: "INVALID_DATE_RANGE",
          message: "End date must be after start date.",
        },
      });
    }

    const allowedTypes = [
      "owner_stay",
      "maintenance",
      "external_booking",
      "manual",
    ];

    if (!allowedTypes.includes(type)) {
      return res.status(400).json({
        success: false,
        error: {
          code: "INVALID_BLOCK_TYPE",
          message: "Choose a valid blocked-date type.",
        },
      });
    }

    const property = await Property.findById(propertyId);

    if (!property) {
      return res.status(404).json({
        success: false,
        error: {
          code: "PROPERTY_NOT_FOUND",
          message: "Property not found.",
        },
      });
    }

    const conflict = await Booking.findOne({
      property: property._id,
      status: "confirmed",
      checkin: { $lt: end },
      checkout: { $gt: start },
    }).select("_id reference");

    if (conflict) {
      return res.status(409).json({
        success: false,

exports.handleStripeWebhook = async (req, res) => {
  const signature = req.headers['stripe-signature']
  const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET

  if (!webhookSecret) {
    console.error('STRIPE_WEBHOOK_SECRET is not configured')
    return res.status(500).send('Webhook is not configured')
  }

  let event

  try {
    event = stripe.webhooks.constructEvent(
      req.body,
      signature,
      webhookSecret
    )
  } catch (error) {
    console.error('Stripe webhook signature verification failed:', error.message)
    return res.status(400).send('Invalid webhook signature')
  }

  try {
    const isSuccessfulEvent =
      event.type === 'checkout.session.completed' ||
      event.type === 'checkout.session.async_payment_succeeded'

    if (!isSuccessfulEvent) {
      return res.status(200).json({ received: true })
    }

    const session = event.data.object

    if (session.payment_status !== 'paid') {
      return res.status(200).json({ received: true })
    }

    const bookingId = session.metadata?.bookingId

    if (!bookingId) {
      console.error('Stripe Checkout Session has no bookingId metadata')
      return res.status(200).json({ received: true })
    }

    const booking = await Booking.findById(bookingId)

    if (!booking) {
      console.error('Booking not found for Stripe session:', session.id)
      return res.status(200).json({ received: true })
    }

    if (booking.payment?.sessionId !== session.id) {
      console.error('Stripe session does not match booking:', booking.reference)
      return res.status(200).json({ received: true })
    }

    if (
      booking.payment?.status === 'paid' &&
      booking.status === 'confirmed'
    ) {
      return res.status(200).json({ received: true })
    }

    if (booking.status !== 'pending') {
      console.error(
        'Paid Stripe session received for a non-pending booking:',
        booking.reference
      )
      return res.status(200).json({ received: true })
    }

    booking.payment.status = 'paid'
    booking.payment.paymentIntentId =
      typeof session.payment_intent === 'string'
        ? session.payment_intent
        : session.payment_intent?.id ?? null
    booking.payment.paidAt = new Date()

    booking.status = 'confirmed'

    await booking.save()

    return res.status(200).json({ received: true })
  } catch (error) {
    console.error('Stripe webhook processing failed:', error)
    return res.status(500).send('Webhook processing failed')
  }
}

function parseDateOnly(value) {
  if (typeof value !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return null;
  }

  const date = new Date(`${value}T00:00:00.000Z`);

  if (
    Number.isNaN(date.getTime()) ||
    date.toISOString().slice(0, 10) !== value
  ) {
    return null;
  }

  return date;
}

async function getQuote(req, res, next) {
  try {
    const { propertyId } = req.params;
    const { checkin, checkout } = req.query;
    const guests = Number(req.query.guests);

    const startDate = parseDateOnly(checkin);
    const endDate = parseDateOnly(checkout);

    if (!startDate || !endDate || endDate <= startDate) {
      return res.status(400).json({
        error: {
          code: "INVALID_DATES",
          message: "Provide valid check-in and check-out dates.",
        },
      });
    }

    if (!Number.isInteger(guests) || guests < 1 || guests > 100) {
      return res.status(400).json({
        error: {
          code: "INVALID_GUEST_COUNT",
          message: "Guests must be a whole number between 1 and 100.",
        },
      });
    }

    const property = await Property.findOne({
      propertyId,
      isActive: true,
      isPlaceholder: false,
      pricingConfigured: true,
    }).lean();

    if (!property) {
      return res.status(404).json({
        error: {
          code: "PROPERTY_NOT_BOOKABLE",
          message: "This property is not configured for booking.",
        },
      });
    }

    if (guests > property.guests) {
      return res.status(400).json({
        error: {
          code: "GUEST_CAPACITY_EXCEEDED",
          message: `This property accommodates a maximum of ${property.guests} guests.`,
        },
      });
    }

    
    const now = new Date();

    const [blockedDate, conflictingBooking] = await Promise.all([
    BlockedDate.findOne({
        property: property._id,
        startDate: { $lt: endDate },
        endDate: { $gt: startDate },
    })
        .select("_id type")
        .lean(),

    Booking.findOne({
        property: property._id,
        checkin: { $lt: endDate },
        checkout: { $gt: startDate },
        $or: [
        { status: "confirmed" },
        {
            status: "pending",
            holdExpiresAt: { $gt: now },
        },
        ],
    })
        .select("_id status")
        .lean(),
    ]);

    if (blockedDate || conflictingBooking) {
    return res.status(409).json({
        error: {
        code: "DATES_UNAVAILABLE",
        message: "This property is not available for the selected dates.",
        },
    });
    }


    const { nights, priceBreakdown } = calculateQuote(
      property,
      startDate,
      endDate,
      guests
    );

    return res.status(200).json({
      data: {
        propertyId,
        checkin,
        checkout,
        guests,
        nights,
        priceBreakdown,
      },
    });
  } catch (error) {
    if (
      error.message === "This property is not configured for booking." ||
      error.message === "A valid nightly rate has not been configured." ||
      error.message === "The cleaning fee configuration is invalid." ||
      error.message === "The minimum-night configuration is invalid." ||
      error.message === "The guest count is invalid for this property." ||
      error.message === "Check-out must be after check-in." ||
      error.message.startsWith("This property requires a minimum stay") ||
      error.message.startsWith("A nightly rate configuration is invalid.") ||
      error.message.startsWith("Unsupported fee calculation")
    ) {
      return res.status(400).json({
        error: {
          code: "QUOTE_VALIDATION_FAILED",
          message: error.message,
        },
      });
    }

    return next(error);
  }
}

const getProperties = async (req, res, next) => {
  try {

    const limit = Number(req.query.limit || 12);

      page < 1 ||
      !Number.isInteger(limit) ||

    ) {
      return res.status(400).json({
        error: {
          code: "INVALID_PAGINATION",
          message: "Page must be positive and limit must be between 1 and 24."
        }
      });
    }

    const filter = {
      isActive: true,
      isPlaceholder: false
    };

    if (req.query.guests !== undefined) {
      const guests = Number(req.query.guests);

      if (!Number.isInteger(guests) || guests < 1) {
        return res.status(400).json({
          error: {
            code: "INVALID_GUESTS",
            message: "Guests must be a positive integer."
          }
        });
      }

      filter.guests = { $gte: guests };
    }

    if (req.query.bedrooms !== undefined) {
      const bedrooms = Number(req.query.bedrooms);

      if (!Number.isInteger(bedrooms) || bedrooms < 0) {
        return res.status(400).json({
          error: {
            code: "INVALID_BEDROOMS",
            message: "Bedrooms must be a non-negative integer."
          }
        });
      }

      filter.bedrooms = { $gte: bedrooms };
    }

    if (req.query.featured !== undefined) {
      if (!["true", "false"].includes(req.query.featured)) {
        return res.status(400).json({
          error: {
            code: "INVALID_FEATURED",
            message: "Featured must be true or false."
          }
        });
      }

      filter.isFeatured = req.query.featured === "true";
    }

    const sortOptions = {
      recommended: { sortOrder: 1, createdAt: -1 },
      rating: { rating: -1, sortOrder: 1 },
      newest: { createdAt: -1 }
    };

    const sortName = req.query.sort || "recommended";

    if (!sortOptions[sortName]) {
      return res.status(400).json({
        error: {
          code: "INVALID_SORT",
          message: "Sort must be recommended, rating, or newest."
        }
      });
    }

    const skip = (page - 1) * limit;

    const [properties, total] = await Promise.all([
      Property.find(filter)
        .select("-createdBy -updatedBy -__v")
        .sort(sortOptions[sortName])
        .skip(skip)
        .limit(limit)
        .lean(false),

      Property.countDocuments(filter)
    ]);

    return res.status(200).json({
      data: properties,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
        hasNextPage: page * limit < total,
        hasPreviousPage: page > 1
      }
    });
  } catch (error) {
    next(error);
  }
};

const jwt = require("jsonwebtoken");
const AdminUser = require("../models/AdminUser");

exports.requireAdmin = async (req, res, next) => {
  try {
    const header = req.headers.authorization;

    if (!header || !header.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        error: {
          code: "AUTHENTICATION_REQUIRED",
          message: "Admin authentication is required.",
        },
      });
    }

    if (!process.env.ADMIN_JWT_SECRET) {
      throw new Error("ADMIN_JWT_SECRET is not configured.");
    }

    const token = header.slice(7);

    const payload = jwt.verify(token, process.env.ADMIN_JWT_SECRET, {
      issuer: "avanta-living-api",
      audience: "avanta-living-admin",
    });

    if (payload.role !== "admin" || !payload.sub) {
      return res.status(401).json({
        success: false,
        error: {
          code: "INVALID_ADMIN_TOKEN",
          message: "Invalid admin token.",
        },
      });
    }

    const admin = await AdminUser.findOne({
      _id: payload.sub,
      role: "admin",
      isActive: true,
    }).select("_id name email role");

    if (!admin) {
      return res.status(401).json({
        success: false,
        error: {
          code: "ADMIN_UNAVAILABLE",
          message: "Admin account is unavailable.",
        },
      });
    }

    req.admin = admin;
    next();
  } catch (error) {
    if (
      error.name === "JsonWebTokenError" ||
      error.name === "TokenExpiredError" ||
      error.name === "NotBeforeError"
    ) {
      return res.status(401).json({
        success: false,
        error: {
          code: "INVALID_ADMIN_TOKEN",
          message: "Admin token is invalid or expired.",
        },
      });
    }

    next(error);
  }
};

function notFoundHandler(req, res, next) {
  res.status(404).json({
    error: {
      code: "ROUTE_NOT_FOUND",
      message: `Route ${req.method} ${req.originalUrl} was not found.`,
    },
  });
}

function errorHandler(err, req, res, next) {
  const statusCode = err.statusCode || err.status || 500;

  if (statusCode >= 500) {
    console.error(err);
  }

  res.status(statusCode).json({
    error: {
      code: err.code || "INTERNAL_SERVER_ERROR",
      message:
        statusCode >= 500
          ? "An unexpected server error occurred."
          : err.message || "Request failed.",
      ...(err.fields ? { fields: err.fields } : {}),
    },
  });
}

module.exports = {
  notFoundHandler,
  errorHandler,
};

require("dotenv").config();

const app = require("./app");
const { connectDatabase } = require("./config/database");

const PORT = Number(process.env.PORT) || 5000;

async function startServer() {
  try {
    await connectDatabase();
  } catch {
    console.error(
      "The API will not start because the configured database connection failed."
    );
    process.exit(1);
  }

  const server = app.listen(PORT, () => {
    console.log(`Avanta Living API running at http://localhost:${PORT}`);
    console.log(`Health check: http://localhost:${PORT}/api/health`);
  });

  function shutdown(signal) {
    console.log(`${signal} received. Shutting down...`);

    server.close(() => {
      console.log("HTTP server closed.");
      process.exit(0);
    });

    setTimeout(() => {
      console.error("Graceful shutdown timed out.");
      process.exit(1);
    }, 10000).unref();
  }

  process.on("SIGINT", () => shutdown("SIGINT"));
  process.on("SIGTERM", () => shutdown("SIGTERM"));
}

startServer();
